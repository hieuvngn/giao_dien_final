   248→export default function MistakeHub() {
   249→  const { toast } = useToast()
   250→  const [tab, setTab] = useState<TabId>('analyze')
   251→  const [records, setRecords] = useState<MistakeRecord[]>([])
   252→  const [input, setInput] = useState('')
   253→  const [analysis, setAnalysis] = useState<AnalysisResult | null>(null)
   254→  const [thinking, setThinking] = useState(false)
   255→  const [filter, setFilter] = useState<MistakeType | 'all'>('all')
   256→  const [q, setQ] = useState('')
   257→  const [confirmDelId, setConfirmDelId] = useState<string | null>(null)
   258→  const [confirmReset, setConfirmReset] = useState(false)
   259→
   260→  useEffect(() => {
   261→    // eslint-disable-next-line react-hooks/set-state-in-effect
   262→    setRecords(loadMistakes())
   263→  }, [])
   264→
   265→  const persist = (next: MistakeRecord[]) => {
   266→    setRecords(next)
   267→    saveMistakes(next)
   268→  }
   269→
   270→  const runAnalyze = (text: string) => {
   271→    const t = text.trim()
   272→    if (!t) {
   273→      toast({ title: '✍️ Chưa có câu nào', description: 'Nhập một câu tiếng Trung, ví dụ: 我昨天去学校了没有。' })
   274→      return
   275→    }
   276→    if (!/[\u4e00-\u9fff]/.test(t)) {
   277→      toast({ title: '🀄 Câu chưa có chữ Hán', description: 'Hãy viết bằng tiếng Trung để AI đối chiếu quy luật lỗi.' })
   278→      return
   279→    }
   280→    setInput(t)
   281→    setAnalysis(null)
   282→    setThinking(true)
   283→    window.setTimeout(() => {
   284→      setAnalysis(analyzeSentence(t))
   285→      setThinking(false)
   286→    }, 650)
   287→  }
   288→
   289→  const saveIssues = () => {
   290→    if (!analysis || analysis.issues.length === 0) return
   291→    const now = Date.now()
   292→    const recs: MistakeRecord[] = analysis.issues.map((it, i) => ({
   293→      id: `${genId()}-${i}`,
   294→      raw: it.raw,
   295→      corrected: it.corrected,
   296→      type: it.type,
   297→      topic: it.topic,
   298→      issue: it.issue,
   299→      explain: it.explain,
   300→      knowledge: it.knowledge,
   301→      level: it.level,
   302→      createdAt: now,
   303→      resolved: false,
   304→      source: 'ai',
   305→    }))
   306→    persist([...recs, ...records])
   307→    toast({ title: `💾 Đã lưu ${recs.length} lỗi vào 错题本`, description: 'Tab "Quy luật của bạn" sẽ cập nhật ngay.' })
   308→    setAnalysis(null)
   309→    setInput('')
   310→  }
   311→
   312→  const toggleResolved = (id: string) => {
   313→    const next = records.map((r) => (r.id === id ? { ...r, resolved: !r.resolved } : r))
   314→    persist(next)
   315→    const rec = next.find((r) => r.id === id)
   316→    if (rec?.resolved) toast({ title: '🎉 Giỏi lắm! Một lỗi đã được khắc phục', description: 'Lỗi này sẽ không còn tính vào quy luật.' })
   317→  }
   318→
   319→  const removeRecord = (id: string) => {
   320→    persist(records.filter((r) => r.id !== id))
   321→    setConfirmDelId(null)
   322→    toast({ title: '🗑 Đã xoá lỗi khỏi sổ' })
   323→  }
   324→
   325→  const resetDemo = () => {
   326→    persist(buildSeedRecords())
   327→    setConfirmReset(false)
   328→    toast({ title: '🔄 Đã nạp lại dữ liệu demo', description: '14 lỗi mẫu: 7 lỗi 了 · 4 lỗi lượng từ · 3 lỗi trật tự.' })
   329→  }
   330→
   331→  const conclusion = buildConclusion(records)
   332→  const openCount = conclusion.open
   333→  const patternTopics = [...new Set(ERROR_PATTERNS.map((p) => p.topic))]
   334→
   335→  const filtered = records
   336→    .filter((r) => (filter === 'all' ? true : r.type === filter))
   337→    .filter((r) => {
   338→      const needle = normSearch(q.trim())
   339→      if (!needle) return true
   340→      return normSearch(`${r.raw} ${r.corrected} ${r.topic} ${r.knowledge.join(' ')} ${r.issue}`).includes(needle)
   341→    })
   342→    .sort((a, b) => b.createdAt - a.createdAt)
   343→  const shown = filtered.slice(0, 30)
   344→
   345→  return (
   346→    <div className="flex min-h-screen flex-col bg-[#fff6e9] text-[#5d4a57]">
   347→      <main className="relative flex-1 pb-16 pt-20 sm:pt-24">
   348→        <div aria-hidden="true" className="dot-grid absolute inset-0" />
   349→
   350→        <div className="relative mx-auto max-w-5xl px-4 sm:px-6">
   351→          {/* ================= header ================= */}
   352→          <Reveal>
   353→            <header className="paper-scrap relative rounded-3xl px-5 py-6 sm:px-8 sm:py-7">
   354→              <WashiTape color="var(--pastel-lilac)" pattern="dots" rotate={-5} className="-top-3 left-10 h-6 w-20" />
   355→              <PushPin color="#9a86c9" className="absolute right-8 top-3 scale-75" />
   356→
   357→              <div className="flex items-start justify-between gap-4">
   358→                <div>
   359→                  <h1 className="leading-tight">
   360→                    <span className="relative inline-block px-3 py-0.5">
   361→                      <span aria-hidden="true" className="absolute inset-0 -rotate-1 rounded-md bg-[var(--pastel-lilac)] opacity-80" />
   362→                      <span className="relative font-display text-2xl font-bold text-[#4f3d4a] sm:text-3xl">Phân tích lỗi sai cá nhân</span>
   363→                      <span className="relative ml-2 hidden font-cn text-2xl text-[#6d5a94] sm:inline" aria-hidden="true">
   364→                        错题本
   365→                      </span>
   366→                    </span>
   367→                  </h1>
   368→                  <p className="mt-2 max-w-xl text-sm font-semibold leading-relaxed text-[#8d7484]">
   369→                    Không chỉ báo đúng/sai — AI <b className="text-[#6d5a94]">phân loại lỗi</b> (ngữ pháp, trợ từ, lượng từ, trật tự từ),
   370→                    <b className="text-[#6d5a94]"> lưu lại lịch sử</b> rồi tìm ra <b className="text-[#6d5a94]">quy luật sai</b> của bạn qua nhiều bài.
   371→                  </p>
   372→                </div>
   373→                <span aria-hidden="true" className="hidden shrink-0 text-5xl sm:block">🧠</span>
   374→              </div>
   375→
   376→              <div className="absolute -bottom-5 right-6 hidden sm:block">
   377→                <PandaSticker className="h-16 w-16" />
   378→              </div>
   379→            </header>
   380→          </Reveal>
   381→
   382→          {/* ================= tabs ================= */}
   383→          <Reveal delay={0.05}>
   384→            <div className="mt-5 flex flex-wrap items-center justify-between gap-3">
   385→              <div className="flex gap-1.5 overflow-x-auto rounded-2xl bg-white/60 p-1.5 ring-1 ring-[#ead9cd]" role="tablist" aria-label="Khu vực sổ lỗi sai">
   386→                {TABS.map((t) => (
   387→                  <button
   388→                    key={t.id}
   389→                    type="button"
   390→                    role="tab"
   391→                    aria-selected={tab === t.id}
   392→                    onClick={() => setTab(t.id)}
   393→                    className={`flex shrink-0 items-center gap-1.5 rounded-xl px-3.5 py-2 text-xs font-extrabold transition-all ${
   394→                      tab === t.id
   395→                        ? 'bg-[#4f3d4a] text-white shadow-sm'
   396→                        : 'text-[#6f5a68] hover:bg-[var(--pastel-butter)]/50'
   397→                    }`}
   398→                  >
   399→                    <span aria-hidden="true">{t.emoji}</span>
   400→                    {t.label}
   401→                    <span className="font-cn text-[11px] opacity-70">{t.cn}</span>
   402→                    {t.id === 'book' && openCount > 0 && (
   403→                      <span className={`rounded-full px-1.5 py-0.5 text-[9px] font-extrabold ${tab === t.id ? 'bg-white/20 text-white' : 'bg-[var(--pastel-pink)] text-[#8a4a63]'}`}>
   404→                        {openCount}
   405→                      </span>
   406→                    )}
   407→                  </button>
   408→                ))}
   409→              </div>
   410→              <p className="text-[10px] font-semibold text-[#b58fa0]">
   411→                {records.length} lỗi trong sổ · {openCount} chưa khắc phục · lưu trên máy bạn
   412→              </p>
   413→            </div>
   414→          </Reveal>
   415→
   416→          <AnimatePresence mode="wait">
   417→            {/* ================= tab 1: phân tích câu ================= */}
   418→            {tab === 'analyze' && (
   419→              <motion.section
   420→                key="analyze"
   421→                role="tabpanel"
   422→                aria-label="Phân tích câu"
   423→                initial={{ opacity: 0, y: 10 }}
   424→                animate={{ opacity: 1, y: 0 }}
   425→                exit={{ opacity: 0, y: -8 }}
   426→                transition={{ duration: 0.25 }}
   427→                className="mt-4 space-y-3"
   428→              >
   429→                <p className="text-[11px] font-bold text-[#8d7484]">
